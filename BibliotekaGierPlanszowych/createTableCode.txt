CREATE TABLE [board_game] (
	id_board_game integer PRIMARY KEY, 
	title varchar,
	id_user integer,
	min_players integer,
	max_players integer,
	rate integer,
	add_date string,
	id_category integer,
	yearpublished integer,
	playingtime integer,
	minplaytime integer,
	maxplaytime integer,
	age integer,
	lastPlayed string,
	loaned integer,
	image_url string
);

CREATE TABLE [category] (
	id_category integer PRIMARY KEY, 
	title_category varchar
);

CREATE TABLE [loaned] (
	id_loaned integer PRIMARY KEY, 
	id_user integer,
	person varchar,
	id_board_game integer
);

CREATE TABLE [wishlist] (
	id_wishlist integer PRIMARY KEY, 
	id_user integer,
	title_wishlist varchar,
	price_wishlist integer
);

CREATE TABLE [users] (
	id_users integer PRIMARY KEY, 
	login varchar UNIQUE,
	password varchar,
	lastUsed integer
);
